-- Khởi tạo thư viện UI RedzLib
local redzlib = loadstring(game:HttpGet("https://raw.githubusercontent.com/daucobonhi/UiRedzV5/refs/heads/main/DemoUi.lua"))()

-- Tạo Cửa sổ giao diện chính
local Windows = redzlib:MakeWindow({
    Title = "PORHub",
    SubTitle = "Make",
    SaveFolder = "tên map"
})

-- Thêm nút thu nhỏ giao diện
Windows:AddMinimizeButton({
    Button = { Image = "rbxassetid://125565119585176", BackgroundTransparency = 0 },
    Corner = { CornerRadius = UDim.new(0, 6) }
})

-- Tạo Tab chức năng chính
local Farm = Windows:MakeTab({"POR.top1", "Info"})

-- ====================================================================
-- CÁC CHỨC NĂNG TRONG TAB "POR.top1"
-- ====================================================================

-- 1. Nút tải Script Chọn chức năng
Farm:AddButton({
    Name = "Bật Chọn Chức Năng",
    Callback = function()
        print("Đang tải Script Chọn chức năng...")
        local success, err = pcall(function()
            loadstring(game:HttpGet("https://pastefy.app/6CKetcSN/raw", true))()
        end)
        if not success then
            warn("Lỗi tải Script Chọn chức năng: " .. tostring(err))
        end
    end
})

-- 2. Nút tải Script Auto Farm
Farm:AddButton({
    Name = "Bật Auto Farm",
    Callback = function()
        print("Đang tải Script Auto Farm...")
        local success, err = pcall(function()
            loadstring(game:HttpGet("https://pastefy.app/BNB7Zj6G/raw", true))()
        end)
        if not success then
            warn("Lỗi tải Script Auto Farm: " .. tostring(err))
        end
    end
})
